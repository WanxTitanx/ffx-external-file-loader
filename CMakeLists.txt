cmake_minimum_required(VERSION 3.20)
project(ffx_external_file_loader VERSION 1.1.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(FF10_BUILD_PRIVATE_RT1 "Build the isolated harness; private PE fixtures are supplied separately" OFF)
add_executable(profile_tests tests/profiles_test.cpp)
add_test(NAME executable_profiles COMMAND profile_tests)
if(WIN32)
  if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Build the runtime with MSVC and -A Win32")
  endif()
  add_library(ff10-file-loader SHARED ff10-file-loader/dllmain.cpp
      ff10-file-loader/hook.cpp ff10-file-loader/module.cpp
      ff10-file-loader/profile_runtime.cpp ff10-file-loader/module.def
      ff10-file-loader/ff10-file-loader.rc)
  target_link_libraries(ff10-file-loader PRIVATE
      "${CMAKE_CURRENT_SOURCE_DIR}/ff10-file-loader/external/simpleLog.lib" bcrypt)
  target_compile_definitions(ff10-file-loader PRIVATE UNICODE _UNICODE NOMINMAX)
  target_compile_options(ff10-file-loader PRIVATE /W4 /WX /utf-8)
  set_target_properties(ff10-file-loader PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  if(FF10_BUILD_PRIVATE_RT1)
    add_executable(native_rt1 tests/native_rt1.cpp ff10-file-loader/hook.cpp ff10-file-loader/profile_runtime.cpp)
    target_link_libraries(native_rt1 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/ff10-file-loader/external/simpleLog.lib" bcrypt)
    target_compile_definitions(native_rt1 PRIVATE UNICODE _UNICODE NOMINMAX FF10_LOADER_TESTING)
    target_compile_options(native_rt1 PRIVATE /W4 /WX /utf-8)
    set_target_properties(native_rt1 PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  endif()
endif()
